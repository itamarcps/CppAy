cmake_minimum_required(VERSION 3.24)
project(AyPlayer VERSION 0.1.0 LANGUAGES CXX)
set(CPPAY_RELEASE_VERSION "${PROJECT_VERSION}")
include(GNUInstallDirs)
install(FILES LICENSE THIRD_PARTY.md reference/AY_EMUL_NOTICE.txt
 DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/cppay)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "C++Ay requires a 64-bit compiler/toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AYPLAYER_GUI "Build Qt Quick desktop player" ON)
option(AYPLAYER_MEMORY_BASELINE "Benchmark original Qt GPU defaults (diagnostic builds only)" OFF)
option(AYPLAYER_SANITIZERS "Enable address/undefined sanitizers" OFF)
option(AYPLAYER_GOLDEN_TESTS "Test locally supplied samples/Flexo02.pt3 and Flexo02.wav" OFF)
if(AYPLAYER_SANITIZERS AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(aycore src/core/engine.cpp src/core/pt3.cpp src/core/logs.cpp)
target_include_directories(aycore PUBLIC src/core)
target_compile_definitions(aycore PUBLIC CPPAY_VERSION="${CPPAY_RELEASE_VERSION}")
if(NOT MSVC)
 target_compile_options(aycore PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(aytool src/core/cli.cpp)
target_link_libraries(aytool PRIVATE aycore)
if(MINGW)
 target_link_options(aytool PRIVATE -municode)
endif()
include(CTest)
if(AYPLAYER_GUI)
 find_package(Qt6 6.8 REQUIRED COMPONENTS Core Quick QuickControls2 Multimedia Concurrent Network Widgets)
 qt_standard_project_setup(REQUIRES 6.8)
 qt_add_executable(AyPlayer src/gui/main.cpp src/gui/controller.cpp src/gui/controller.h src/gui/playlistmodel.cpp src/gui/playlistmodel.h src/gui/playbacksession.cpp src/gui/playbacksession.h)
 if(AYPLAYER_MEMORY_BASELINE)
  target_compile_definitions(AyPlayer PRIVATE AYPLAYER_MEMORY_BASELINE)
 endif()
 qt_add_qml_module(AyPlayer URI AyPlayer VERSION 1.0 OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/qml/AyPlayer QML_FILES qml/Main.qml qml/RetroButton.qml qml/RetroSlider.qml qml/RetroPanel.qml qml/RetroCheckBox.qml qml/RetroComboBox.qml qml/RetroSpinBox.qml qml/WindowTitleBar.qml qml/WindowResizeHandles.qml)
 qt_add_resources(AyPlayer "app_icon" PREFIX "/" FILES assets/cppay-icon.png assets/cppay-logo.png)
 set_target_properties(AyPlayer PROPERTIES WIN32_EXECUTABLE TRUE OUTPUT_NAME "C++Ay")
 if(WIN32)
  target_sources(AyPlayer PRIVATE packaging/windows/ayplayer.rc)
  target_include_directories(AyPlayer PRIVATE packaging/windows)
 endif()
 target_link_libraries(AyPlayer PRIVATE aycore Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Concurrent Qt6::Network Qt6::Widgets)
 install(TARGETS AyPlayer aytool RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
 if(UNIX)
 install(FILES packaging/cppay.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
 install(FILES assets/cppay-icon.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/64x64/apps RENAME cppay.png)
 install(FILES packaging/ayplayer-mime.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/mime/packages)
 endif()
else()
 install(TARGETS aytool RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(BUILD_TESTING)
 set(INTEGRATION_PT3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/integration.pt3)
 set(INTEGRATION_WAV ${CMAKE_CURRENT_BINARY_DIR}/integration-reference.wav)
 add_executable(core_tests tests/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE aycore)
 add_test(NAME core_behavior COMMAND core_tests ${INTEGRATION_PT3})
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME comparator_behavior COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_comparator.py)
 # This render checks integration consistency, not independent engine accuracy.
 # Immutable source-oracle fixtures are tested separately below.
 add_test(NAME integration_prepare COMMAND ${CMAKE_COMMAND} -E rm -f ${INTEGRATION_WAV})
 set_tests_properties(integration_prepare PROPERTIES FIXTURES_SETUP integration_clean)
 add_test(NAME integration_render COMMAND aytool render ${INTEGRATION_PT3} ${INTEGRATION_WAV})
 set_tests_properties(integration_render PROPERTIES FIXTURES_REQUIRED integration_clean FIXTURES_SETUP integration_audio)
 if(AYPLAYER_GOLDEN_TESTS)
 foreach(sample Flexo02.pt3 Flexo02.wav)
  if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/samples/${sample})
   message(FATAL_ERROR "AYPLAYER_GOLDEN_TESTS requires locally supplied samples/${sample}; these files are not distributed")
  endif()
 endforeach()
 add_test(NAME golden_render COMMAND aytool render ${CMAKE_CURRENT_SOURCE_DIR}/samples/Flexo02.pt3 ${CMAKE_CURRENT_BINARY_DIR}/golden-candidate.wav)
 # Render refuses overwrites; remove only this generated test artifact before each test run.
 add_test(NAME golden_prepare COMMAND ${CMAKE_COMMAND} -E rm -f ${CMAKE_CURRENT_BINARY_DIR}/golden-candidate.wav)
 set_tests_properties(golden_prepare PROPERTIES FIXTURES_SETUP golden_clean)
 set_tests_properties(golden_render PROPERTIES FIXTURES_REQUIRED golden_clean FIXTURES_SETUP golden_candidate)
 add_test(NAME golden_pcm COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/compare_pcm.py ${CMAKE_CURRENT_SOURCE_DIR}/samples/Flexo02.wav ${CMAKE_CURRENT_BINARY_DIR}/golden-candidate.wav --report ${CMAKE_CURRENT_BINARY_DIR}/golden-comparison.json)
 set_tests_properties(golden_pcm PROPERTIES FIXTURES_REQUIRED golden_candidate)
 endif()
endif()

 add_executable(gui_controller_tests tests/gui_controller_tests.cpp src/gui/controller.cpp src/gui/controller.h src/gui/playlistmodel.cpp src/gui/playlistmodel.h src/gui/playbacksession.cpp src/gui/playbacksession.h)

 add_test(NAME independent_source_fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reference_fixtures.py $<TARGET_FILE:aytool>)
